package AccesoADatos;

/**
 * Maneja la persistencia de registros del parqueo en Archivos.txt
 * Formato de cada línea: placa;tipo;horaEntrada;horaSalida;monto
 * horaSalida y monto quedan como "null" y "0.0" mientras el vehículo está activo.
 * @author Ignacio CH
 */
import Entidades.Vehiculo;
import java.io.*;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.List;

public class CrearArchivo {

    private static final String ARCHIVO   = "Archivos.txt";
    private static final String SEPARADOR = ";";

    /** Escribe TODA la lista al archivo (sobreescribe). */
    public void guardarTodos(List<Vehiculo> vehiculos) {
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(ARCHIVO, false))) {
            for (Vehiculo v : vehiculos) {
                String salida = (v.getHoraSalida() == null) ? "null"
                                                            : v.getHoraSalida().toString();
                bw.write(
                    v.getPlaca()            + SEPARADOR +
                    v.getTipo()             + SEPARADOR +
                    v.getHoraEntrada()      + SEPARADOR +
                    salida                  + SEPARADOR +
                    v.getMonto()
                );
                bw.newLine();
            }
        } catch (IOException e) {
            System.err.println("Error al guardar archivo: " + e.getMessage());
        }
    }

    /** Lee el archivo y reconstruye la lista de vehículos. */
    public List<Vehiculo> leerTodos() {
        List<Vehiculo> lista = new ArrayList<>();
        File archivo = new File(ARCHIVO);

        if (!archivo.exists()) return lista;

        try (BufferedReader br = new BufferedReader(new FileReader(ARCHIVO))) {
            String linea;
            while ((linea = br.readLine()) != null) {
                if (linea.trim().isEmpty()) continue;

                String[] p = linea.split(SEPARADOR);
                if (p.length == 5) {
                    String    placa      = p[0].trim();
                    String    tipo       = p[1].trim();
                    LocalTime entrada    = LocalTime.parse(p[2].trim());
                    LocalTime salida     = p[3].trim().equals("null") ? null
                                          : LocalTime.parse(p[3].trim());
                    double    monto      = Double.parseDouble(p[4].trim());

                    lista.add(new Vehiculo(placa, tipo, entrada, salida, monto));
                }
            }
        } catch (IOException e) {
            System.err.println("Error al leer archivo: " + e.getMessage());
        }
        return lista;
    }

} // fin clase CrearArchivo

package Logica;

/**
 * Capa de lógica de negocio del sistema de parqueo.
 * Aplica validaciones, reglas de unicidad, cálculo de cobro
 * y coordina con la capa de Acceso a Datos.
 * @author Ignacio CH
 */
import AccesoADatos.CrearArchivo;
import Entidades.Vehiculo;
import java.time.LocalTime;
import java.time.Duration;
import java.util.List;
import java.util.stream.Collectors;

public class GestionParqueo {

    private static final double TARIFA_POR_HORA = 500.0;

    // Patrón: exactamente 6 dígitos  O  3 letras + 3 dígitos (ej: ABC123)
    private static final String PATRON_PLACA =
            "^[0-9]{6}$|^[A-Za-z]{3}[0-9]{3}$";

    private final CrearArchivo dao;
    private List<Vehiculo>     vehiculos; // caché en memoria

    public GestionParqueo() {
        this.dao      = new CrearArchivo();
        this.vehiculos = dao.leerTodos(); // carga inicial desde archivo
    }

    // ── Validaciones ─────────────────────────────────────────────────────────

    /**
     * Valida que la placa cumpla con el formato permitido.
     * @return true si es válida
     */
    public boolean placaValida(String placa) {
        if (placa == null || placa.trim().isEmpty()) return false;
        return placa.trim().toUpperCase().matches(PATRON_PLACA);
    }

    /**
     * Verifica si ya existe un vehículo ACTIVO con esa placa (sin salida).
     * @return true si la placa ya está dentro del parqueo
     */
    public boolean placaActiva(String placa) {
        return vehiculos.stream()
                .anyMatch(v -> v.getPlaca().equalsIgnoreCase(placa.trim())
                            && v.estaActivo());
    }

    // ── Operaciones principales ───────────────────────────────────────────────

    /**
     * Registra la entrada de un vehículo.
     * @param placa Placa del vehículo
     * @param tipo  "Carro" o "Moto"
     * @return Mensaje de resultado de la operación
     */
    public String registrarEntrada(String placa, String tipo) {
        // Validar campos obligatorios
        if (placa == null || placa.trim().isEmpty()) {
            return "ERROR: La placa es obligatoria.";
        }
        if (tipo == null || tipo.trim().isEmpty()) {
            return "ERROR: El tipo de vehículo es obligatorio.";
        }

        placa = placa.trim().toUpperCase();

        // Validar formato de placa (Regla 3)
        if (!placaValida(placa)) {
            return "ERROR: Formato de placa inválido. "
                 + "Debe ser 6 dígitos (123456) o 3 letras + 3 números (ABC123).";
        }

        // Validar unicidad (Regla 1)
        if (placaActiva(placa)) {
            return "ERROR: El vehículo con placa " + placa
                 + " ya se encuentra en el parqueo.";
        }

        // Registrar con hora automática
        Vehiculo v = new Vehiculo(placa, tipo.trim(), LocalTime.now());
        vehiculos.add(v);
        dao.guardarTodos(vehiculos);

        return "OK: Entrada registrada para " + placa
             + " (" + tipo + ") a las " + v.getHoraEntrada() + ".";
    }

    /**
     * Registra la salida de un vehículo y calcula el monto.
     * @param placa Placa del vehículo a retirar
     * @return Mensaje con el monto a cobrar o error
     */
    public String registrarSalida(String placa) {
        if (placa == null || placa.trim().isEmpty()) {
            return "ERROR: La placa es obligatoria.";
        }

        placa = placa.trim().toUpperCase();

        // Buscar vehículo activo con esa placa
        Vehiculo objetivo = null;
        for (Vehiculo v : vehiculos) {
            if (v.getPlaca().equalsIgnoreCase(placa) && v.estaActivo()) {
                objetivo = v;
                break;
            }
        }

        if (objetivo == null) {
            return "ERROR: No se encontró un vehículo activo con placa " + placa + ".";
        }

        // Registrar salida y calcular cobro (Regla 2)
        LocalTime horaSalida = LocalTime.now();
        double monto = calcularMonto(objetivo.getHoraEntrada(), horaSalida);

        objetivo.setHoraSalida(horaSalida);
        objetivo.setMonto(monto);
        dao.guardarTodos(vehiculos);

        return String.format(
            "OK: Salida de %s a las %s. Tiempo: %s. Monto a pagar: ₡%.0f",
            placa, horaSalida,
            formatearDuracion(objetivo.getHoraEntrada(), horaSalida),
            monto
        );
    }

    /**
     * Elimina un registro del historial (solo vehículos con salida registrada).
     * @param placa Placa del registro a eliminar
     * @return Mensaje de resultado
     */
    public String eliminarHistorial(String placa) {
        if (placa == null || placa.trim().isEmpty()) {
            return "ERROR: La placa es obligatoria.";
        }
        placa = placa.trim().toUpperCase();
        final String placaFinal = placa;

        boolean eliminado = vehiculos.removeIf(
            v -> v.getPlaca().equalsIgnoreCase(placaFinal) && !v.estaActivo()
        );

        if (!eliminado) {
            return "ERROR: No se encontró registro histórico para la placa " + placa + ".";
        }

        dao.guardarTodos(vehiculos);
        return "OK: Registro histórico de " + placa + " eliminado.";
    }

    // ── Consultas para la vista ───────────────────────────────────────────────

    /** Devuelve solo los vehículos actualmente en el parqueo (sin salida). */
    public List<Vehiculo> getVehiculosActivos() {
        return vehiculos.stream()
                .filter(Vehiculo::estaActivo)
                .collect(Collectors.toList());
    }

    /** Devuelve el historial completo (vehículos que ya salieron). */
    public List<Vehiculo> getHistorial() {
        return vehiculos.stream()
                .filter(v -> !v.estaActivo())
                .collect(Collectors.toList());
    }

    // ── Cálculos internos ─────────────────────────────────────────────────────

    /**
     * Calcula el monto a pagar: 500 por hora o fracción (Regla 2).
     */
    private double calcularMonto(LocalTime entrada, LocalTime salida) {
        long minutos = Duration.between(entrada, salida).toMinutes();
        if (minutos < 0) minutos = 0; // por si cruza medianoche (edge case)
        long horas = (long) Math.ceil(minutos / 60.0); // fracción cuenta como hora
        if (horas == 0) horas = 1; // mínimo 1 hora
        return horas * TARIFA_POR_HORA;
    }

    /** Devuelve la duración formateada como "Xh Ym". */
    private String formatearDuracion(LocalTime entrada, LocalTime salida) {
        long minutos = Duration.between(entrada, salida).toMinutes();
        if (minutos < 0) minutos = 0;
        return (minutos / 60) + "h " + (minutos % 60) + "m";
    }

} // fin clase GestionParqueo

2. ¿Cómo funciona ese for con ?
Ese ? no es parte del for, es el operador ternario. Son dos cosas separadas. Miremos la línea en cuestión:
String salida = (v.getHoraSalida() == null) ? "null" : v.getHoraSalida().toString();
Estructura del operador ternario
resultado = (condición) ? valor_si_verdadero : valor_si_falso;

Es exactamente equivalente a escribir este if/else:
String salida;
if (v.getHoraSalida() == null) {
    salida = "null";
} else {
    salida = v.getHoraSalida().toString();
}

¿Por qué se usa aquí?
Cuando un vehículo aún está en el parqueo, horaSalida vale null (no existe aún). Si intentáramos llamar .toString() directamente sobre null, el programa lanzaría un NullPointerException y se caería. El ternario protege contra eso:
Estado del vehículo getHoraSalida() Lo que graba en el archivo
Sigue en parqueo null el texto "null" Ya salióLocalTime (ej: 14:30)"14:30"

Luego al leer el archivo, CrearArchivo hace el proceso inverso:
LocalTime salida = p[3].trim().equals("null") ? null : LocalTime.parse(p[3].trim());

Si encuentra el texto "null" → asigna null. Si encuentra una hora → la convierte a LocalTime.

Para eliminar los nanosegundos (esos números adicionales después de los segundos) en Java, tienes dos caminos: truncar el tiempo al momento de crearlo o darle formato solo al mostrarlo en la tabla.

Aquí tienes las mejores opciones para tu código:
1. Truncar los segundos en la Lógica (GestionParqueo.java)

2. Dar formato en la Interfaz (App.java)

Si prefieres que los datos se guarden con precisión total pero se vean "limpios" en la tabla, debes usar un DateTimeFormatter en tu clase App.

En los métodos de refrescar tablas, haz lo siguiente:
Java

import java.time.format.DateTimeFormatter; // Importar

// Crea un formateador (ejemplo: HH:mm:ss o HH:mm)
private final DateTimeFormatter formateador = DateTimeFormatter.ofPattern("HH:mm:ss");

private void refrescarTablaActivos() {
    modeloActivos.setRowCount(0);
    List<Registro> activos = gestion.getVehiculosActivos();
    for (Registro v : activos) {
        modeloActivos.addRow(new Object[]{
            v.getPlaca(),
            v.getTipo(),
            v.getHoraEntrada().format(formateador) // Aplicar formato aquí
        });
    }
}

private void refrescarTablaHistorial() {
    modeloHistorial.setRowCount(0);
    List<Registro> historial = gestion.getHistorial();
    for (Registro v : historial) {
        // La salida puede ser null, hay que validar antes de formatear
        String salidaFormateada = (v.getHoraSalida() != null) 
                                  ? v.getHoraSalida().format(formateador) 
                                  : "";

        modeloHistorial.addRow(new Object[]{
            v.getPlaca(),
            v.getTipo(),
            v.getHoraEntrada().format(formateador),
            salidaFormateada,
            String.format("%.0f", v.getMonto())
        });
    }
}

¿Cuál elegir?

    Usa la Opción 1 (Truncar) si no necesitas precisión de milisegundos para nada. Hará que tu archivo de texto sea más fácil de leer.

    Usa la Opción 2 (Formatear) si quieres un control total sobre cómo se ve (por ejemplo, si solo quieres mostrar horas y minutos "19:38" sin los segundos).
